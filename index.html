// =============================== // تنظیمات اولیه سایت // =============================== document.addEventListener("DOMContentLoaded", () => { // منوی موبایل const menuButton = document.querySelector("#menuButton"); const menu = document.querySelector("#menu"); if (menuButton && menu) { menuButton.addEventListener("click", () => { menu.classList.toggle("active"); }); } // اسکرول نرم منو const links = document.querySelectorAll("a[href^='#']"); links.forEach(link => { link.addEventListener("click", function(event) { const target = document.querySelector(this.getAttribute("href")); if (target) { event.preventDefault(); target.scrollIntoView({ behavior: "smooth" }); } }); }); // دکمه شروع const startButton = document.querySelector("#startButton"); if (startButton) { startButton.addEventListener("click", () => { const services = document.querySelector("#services"); if (services) { services.scrollIntoView({ behavior: "smooth" }); } }); } // دکمه حالت تاریک const darkButton = document.querySelector("#darkButton"); if (darkButton) { darkButton.addEventListener("click", () => { document.body.classList.toggle("dark"); }); } // انیمیشن کارت‌ها هنگام ورود به صفحه const cards = document.querySelectorAll(".card"); const observer = new IntersectionObserver((entries) => { entries.forEach(entry => { if (entry.isIntersecting) { entry.target.classList.add("show"); } }); }, { threshold: 0.2 }); cards.forEach(card => { observer.observe(card); }); // فرم تماس const contactForm = document.querySelector("#contactForm"); if (contactForm) { contactForm.addEventListener("submit", function(event) { event.preventDefault(); const name = document.querySelector("#name").value; const email = document.querySelector("#email").value; const message = document.querySelector("#message").value; if (!name || !email || !message) { alert("لطفاً همه قسمت‌ها را کامل کنید."); return; } alert("پیام شما با موفقیت ارسال شد ❤️"); contactForm.reset(); }); } // نمایش سال فعلی در فوتر const year = document.querySelector("#year"); if (year) { year.textContent = new Date().getFullYear(); } });
